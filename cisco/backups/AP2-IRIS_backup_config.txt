! Date config : Avril 2026 (Day0 via interface web EWC)
! =============================================================================
!
! ─── ETAT CONFIGURE (Day0 via GUI) ──────────────────────────────────────────
!
!  Nom AP           : AP2-IRIS
!  MAC AP           : 70:BC:48:C8:9F:68
!  IP Management    : 192.168.50.5 / 255.255.255.0 (STATIQUE, VLAN 50)
!  Gateway          : 192.168.50.1
!  Port Switch      : Fa0/21 (SW2-IRIS) — trunk, native VLAN 50
!  SSID             : IRIS (WPA2-Enterprise / 802.1X)
!  RADIUS Server    : 192.168.50.10 — port 1812/1813
!  RADIUS Key       : RadiusAP_IRIS_2026!
!  Admin login      : admin / NVTech_Admin2026!
!  Acces web        : https://192.168.50.5
!  Acces SSH        : ssh admin@192.168.50.5
!
! ─── COMPATIBILITE PROJETS ───────────────────────────────────────────────────
!
!  RP01 actif : RADIUS = NPS sur DC-IRIS-01 (192.168.50.10)
!               Cle NPS : RadiusAP_IRIS_2026! (client AP-IRIS dans NPS)
!
!  RP06 actif : RADIUS = FreeRADIUS dans Docker sur SRV_MAIN (192.168.50.10)
!               Cle FR  : RadiusAP_IRIS_2026! (client AP-IRIS dans clients.conf)
!
! ─── VLANS WIFI — OPTION A (Multi-SSID, VLAN statique par policy) ────────────
!
!  Un SSID dédié par rôle (pas d'aaa-override, VLAN fixe dans la policy) :
!  - IRIS-WIFI  → POLICY-ETUDIANTS → VLAN 10  (802.1X, GRP_Etudiants_SISR/SLAM)
!  - IRIS-PROFS → POLICY-PROFS     → VLAN 20  (802.1X, GRP_Profs)
!  - IRIS-ADMIN → POLICY-ADMIN     → VLAN 30  (802.1X, GRP_Administration)
!  - IRIS-GUEST → POLICY-INVITES   → VLAN 40  (802.1X, GRP_Invites)
!
!  Limitation EWC 17.9 : aaa-override + FlexConnect local switching = VLAN failure
!  Solution : VLAN statique par policy, NPS authentifie sans pousser de VLAN
!
! =============================================================================
! CONFIGURATION CLI COMPLETE (pour restauration ou reference)
! Appliquer via console ou SSH si reset usine necessaire
! =============================================================================

enable
configure terminal

hostname AP2-IRIS

! ── Management ─────────────────────────────────────────────────────
interface Vlan50
 ip address 192.168.50.5 255.255.255.0
 no shutdown

wireless management interface Vlan50
ip default-gateway 192.168.50.1

! ── DNS / NTP ──────────────────────────────────────────────────────
ip domain-name iris.local
ip name-server 192.168.50.1
ntp server 192.168.50.1
clock timezone CET 1 0

! ── Compte admin ───────────────────────────────────────────────────
username admin privilege 15 secret NVTech_Admin2026!
enable secret NVTech_Admin2026!

! ── SSH ────────────────────────────────────────────────────────────
crypto key generate rsa modulus 2048
ip ssh version 2
line vty 0 15
 transport input ssh
 login local
 access-class MGMT-ACL in
ip access-list standard MGMT-ACL
 permit 192.168.50.0 0.0.0.255
 deny any log

! ── Syslog ─────────────────────────────────────────────────────────
logging host 192.168.50.10
logging trap informational
no ip http server
no cdp run

! ── RADIUS / AAA ───────────────────────────────────────────────────
aaa new-model
radius server RADIUS-IRIS-PRIMARY
 address ipv4 192.168.50.10 auth-port 1812 acct-port 1813
 key RadiusAP_IRIS_2026!
 timeout 5
 retransmit 3

aaa group server radius RADIUS-IRIS
 server name RADIUS-IRIS-PRIMARY
 deadtime 10

aaa authentication dot1x RADIUS-DOT1X group RADIUS-IRIS
aaa authorization network RADIUS-DOT1X group RADIUS-IRIS
aaa accounting dot1x RADIUS-DOT1X start-stop group RADIUS-IRIS

! ── Profils VLAN — Option A (VLAN statique, pas d'aaa-override) ────
! NOTE: aaa-override DESACTIVE sur toutes les policies (EWC 17.9 incompatible
!       avec aaa-override + FlexConnect local switching = VLAN failure)

wireless profile policy POLICY-ETUDIANTS
 description "ETUDIANTS"
 vlan 10
 no aaa-override
 no shutdown

wireless profile policy POLICY-PROFS
 description "PROFS"
 vlan 20
 no aaa-override
 no shutdown

wireless profile policy POLICY-ADMIN
 description "ADMINISTRATION"
 vlan 30
 no aaa-override
 no shutdown

wireless profile policy POLICY-INVITES
 description "GUESTS"
 vlan 40
 no aaa-override
 no shutdown

! ── WLANs — 4 SSIDs (un par rôle) ─────────────────────────────────
wlan IRIS-WIFI 1 IRIS-WIFI
 no security wpa akm psk
 security wpa akm dot1x
 security wpa wpa2
 security wpa wpa2 ciphers aes
 security dot1x authentication-list RADIUS-DOT1X
 no shutdown

wlan IRIS-PROFS 2 IRIS-PROFS
 no security wpa akm psk
 security wpa akm dot1x
 security wpa wpa2
 security wpa wpa2 ciphers aes
 security dot1x authentication-list RADIUS-DOT1X
 no shutdown

wlan IRIS-ADMIN 3 IRIS-ADMIN
 no security wpa akm psk
 security wpa akm dot1x
 security wpa wpa2
 security wpa wpa2 ciphers aes
 security dot1x authentication-list RADIUS-DOT1X
 no shutdown

wlan IRIS-GUEST 4 IRIS-GUEST
 no security wpa akm psk
 security wpa akm dot1x
 security wpa wpa2
 security wpa wpa2 ciphers aes
 security dot1x authentication-list RADIUS-DOT1X
 no shutdown

! ── Tags ───────────────────────────────────────────────────────────
wireless tag policy PTAG-IRIS
 wlan IRIS-WIFI policy POLICY-ETUDIANTS
 wlan IRIS-PROFS policy POLICY-PROFS
 wlan IRIS-ADMIN policy POLICY-ADMIN
 wlan IRIS-GUEST policy POLICY-INVITES

wireless tag site SITE-TAG-IRIS
 ap-profile default-ap-profile
 no local-site

! ── AP ─────────────────────────────────────────────────────────────
ap 70BC.48C8.9F68
 name AP2-IRIS
 tag policy PTAG-IRIS
 tag site SITE-TAG-IRIS

! ── RF ─────────────────────────────────────────────────────────────
ap dot11 24ghz txpower global auto
ap dot11 5ghz txpower global auto
ap dot11 24ghz channel global auto
ap dot11 5ghz channel global auto

end
write memory
